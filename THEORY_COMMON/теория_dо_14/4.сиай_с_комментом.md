# ============================================================
#  CI/CD для Bank of Anthos (проект ДО14, Школа 21)
#  4 стадии: build -> test -> staging -> prod
#
#  ДВА РЕЖИМА СЕКРЕТОВ (файл одинаковый для обоих):
#    * Режим "кластер" (сейчас): токен Docker Hub примонтирован
#      к подам заданий секретом dockerhub-auth; ключи JWT лежат
#      секретом jwt-key в namespace staging/prod. Переменные в
#      GitLab НЕ нужны.
#    * Режим "GitLab" (потом): если заданы переменные
#      DOCKERHUB_TOKEN и JWT_PRIVATE_KEY/JWT_PUBLIC_KEY — пайплайн
#      возьмёт их. Достаточно добавить переменные, файл не менять.
#
#  Деплой идёт "изнутри": раннер крутит задания подами в k3s,
#  helm/kubectl используют доступ пода (in-cluster), kubeconfig не нужен.
# ============================================================

stages:
  - build
  - test
  - staging
  - prod

variables:
  IMAGE_TAG: "$CI_COMMIT_SHORT_SHA"   # тег = короткий хэш коммита
  CHART: "src/helm/bank-of-anthos"    # путь к Helm-чарту от корня репозитория
  DOCKERHUB_USER: "ЗАМЕНИ_НА_СВОЙ_ЛОГИН_DOCKERHUB"   # логин — не секрет, можно прямо тут

# ------------------------------------------------------------
#  BUILD: Kaniko собирает образ каждого сервиса и пушит в Docker Hub
# ------------------------------------------------------------
build:
  stage: build
  image:
    name: gcr.io/kaniko-project/executor:debug
    entrypoint: [""]
  parallel:
    matrix:
      - SERVICE: frontend
        CONTEXT: src/bank-of-anthos/src/frontend
      - SERVICE: userservice
        CONTEXT: src/bank-of-anthos/src/accounts/userservice
      - SERVICE: contacts
        CONTEXT: src/bank-of-anthos/src/accounts/contacts
      - SERVICE: accounts-db
        CONTEXT: src/bank-of-anthos/src/accounts/accounts-db
      - SERVICE: ledger-db
        CONTEXT: src/bank-of-anthos/src/ledger/ledger-db
      - SERVICE: balancereader
        CONTEXT: src/bank-of-anthos/src/ledger/balancereader
      - SERVICE: ledgerwriter
        CONTEXT: src/bank-of-anthos/src/ledger/ledgerwriter
      - SERVICE: transactionhistory
        CONTEXT: src/bank-of-anthos/src/ledger/transactionhistory
  script:
    - |
      if [ -n "$DOCKERHUB_TOKEN" ]; then
        echo "Режим GitLab-переменных: создаю config.json из токена."
        mkdir -p /kaniko/.docker
        AUTH=$(printf "%s:%s" "$DOCKERHUB_USER" "$DOCKERHUB_TOKEN" | base64 | tr -d '\n')
        echo "{\"auths\":{\"https://index.docker.io/v1/\":{\"auth\":\"$AUTH\"}}}" > /kaniko/.docker/config.json
      else
        echo "Режим кластерного секрета: использую примонтированный /kaniko/.docker/config.json."
      fi
    - >-
      /kaniko/executor
      --context "$CI_PROJECT_DIR/$CONTEXT"
      --dockerfile "$CI_PROJECT_DIR/$CONTEXT/Dockerfile"
      --destination "docker.io/$DOCKERHUB_USER/$SERVICE:$IMAGE_TAG"
      --destination "docker.io/$DOCKERHUB_USER/$SERVICE:latest"

# ------------------------------------------------------------
#  Шаблон деплоя (helm). NS/HOST задаются в каждой стадии.
#  JWT-ключи добавляются только если заданы переменные (режим GitLab);
#  иначе используется заранее созданный секрет jwt-key в namespace.
# ------------------------------------------------------------
.deploy:
  image: alpine/k8s:1.30.2          # внутри есть helm и kubectl
  script:
    - |
      JWT_ARGS=""
      if [ -n "$JWT_PRIVATE_KEY" ] && [ -n "$JWT_PUBLIC_KEY" ]; then
        JWT_ARGS="--set-file jwt.privateKey=$JWT_PRIVATE_KEY --set-file jwt.publicKey=$JWT_PUBLIC_KEY"
      fi
    - >-
      helm upgrade --install boa "$CHART"
      -n "$NS"
      --set imageRepo="docker.io/$DOCKERHUB_USER"
      --set imageTag="$IMAGE_TAG"
      --set appVersion="$IMAGE_TAG"
      --set ingress.host="$HOST"
      $JWT_ARGS
      --wait --timeout 5m
    - kubectl get pods -n "$NS"

deploy-staging:
  extends: .deploy
  stage: staging
  variables:
    NS: staging
    HOST: bank-staging.10.174.8.95.nip.io
  environment:
    name: staging
    url: http://bank-staging.10.174.8.95.nip.io

deploy-prod:
  extends: .deploy
  stage: prod
  variables:
    NS: prod
    HOST: bank.10.174.8.95.nip.io
  environment:
    name: prod
    url: http://bank.10.174.8.95.nip.io
  when: manual           # запускается только кнопкой в GitLab
